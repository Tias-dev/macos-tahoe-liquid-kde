cmake_minimum_required(VERSION 3.22)
project(MenuVisualProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus)
find_package(KWayland CONFIG REQUIRED)
find_package(KF6Svg CONFIG REQUIRED)
find_package(Wayland REQUIRED COMPONENTS Client)
add_executable(menu-wayland-pointer wayland_pointer.cpp)
target_link_libraries(menu-wayland-pointer PRIVATE Qt6::Core Qt6::Gui Qt6::DBus Plasma::KWaylandClient Wayland::Client)
add_executable(menu-svg-frames svg_frames.cpp)
target_link_libraries(menu-svg-frames PRIVATE Qt6::Gui KF6::Svg)
